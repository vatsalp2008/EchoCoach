# ── Neural TTS (Kokoro-82M; optional; see app/tts.py) ───────────────────────
# Without these, the interviewer just uses the browser's built-in voice —
# nothing breaks. Two engines, same model and voice:
#   • Apple Silicon: mlx-audio (MLX), the team's Macs.
#   • Linux (the Cloud Run image): kokoro-onnx on ONNX Runtime, CPU. Its model
#     files are downloaded by the Dockerfile, not pip.
#
# Kept out of requirements.txt because misaki's metadata still caps
# Requires-Python at <3.13 even though it runs fine on 3.14 (verified), so on a
# Mac with Python 3.14 pip needs a flag we don't want applied to the main install:
#
#   .venv/bin/pip install --ignore-requires-python -r requirements-tts.txt
#
# (The container runs Python 3.12, so it needs no flag.)
#
# Pinned to a set that installs from wheels alongside cognee/litellm (which pin
# click and tokenizers exactly); left unpinned, pip backtracks into spaCy
# versions that must be compiled from source and fail on 3.14.
mlx-audio==0.5.6 ; sys_platform == "darwin" and platform_machine == "arm64"
misaki[en]==0.9.4 ; sys_platform == "darwin" and platform_machine == "arm64"
spacy==3.8.13 ; sys_platform == "darwin" and platform_machine == "arm64"
spacy-curated-transformers==0.3.1 ; sys_platform == "darwin" and platform_machine == "arm64"
transformers==5.15.1 ; sys_platform == "darwin" and platform_machine == "arm64"
